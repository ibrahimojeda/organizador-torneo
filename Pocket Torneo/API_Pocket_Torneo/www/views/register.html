' + esc(c.name) + '
' + '
' + (c.dojo ? esc(c.dojo) : 'Sin dojo') + ' ' + cl + '
' + '
' + '' + '' + '
'; }).join(''); } window.editComp = function(id) { var comps = DB.getCompetitors(); for (var i = 0; i < comps.length; i++) { if (comps[i].id === id) { document.getElementById('comp-name').value = comps[i].name; document.getElementById('comp-dojo').value = comps[i].dojo || ''; document.getElementById('comp-category').value = comps[i].category || 'kata'; editingCompId = id; document.getElementById('btn-add-comp').textContent = '✏️ Actualizar'; break; } } }; window.deleteComp = function(id) { if (confirm('¿Eliminar este competidor?')) { DB.deleteCompetitor(id); renderCompetitors(); showToast('🗑️ Competidor eliminado'); } }; function addComp() { var name = document.getElementById('comp-name').value.trim(); var dojo = document.getElementById('comp-dojo').value; var cat = document.getElementById('comp-category').value; if (!name) { showToast('⚠️ Escribe un nombre'); return; } if (editingCompId) { DB.updateCompetitor(editingCompId, name, dojo, cat); editingCompId = null; document.getElementById('btn-add-comp').textContent = '+ Añadir Competidor'; showToast('✏️ Competidor actualizado', 'success'); } else { DB.addCompetitor(name, dojo, cat); showToast('✅ Competidor añadido', 'success'); } document.getElementById('comp-name').value = ''; renderCompetitors(); } document.getElementById('btn-add-comp').addEventListener('click', addComp); document.getElementById('comp-name').addEventListener('keydown', function(e) { if (e.key === 'Enter') { e.preventDefault(); addComp(); } }); function esc(str) { return (str || '').replace(/&/g,'&').replace(//g,'>').replace(/"/g,'"'); } function renderAll() { renderDojos(); renderCompetitors(); } function showToast(msg, type) { var e = document.querySelector('.toast'); if (e) e.remove(); var el = document.createElement('div'); el.className = 'toast' + (type === 'success' ?
